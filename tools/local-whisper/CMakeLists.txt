cmake_minimum_required(VERSION 3.16)
project(mimi_whisper_worker LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(GGML_METAL ON CACHE BOOL "" FORCE)
set(GGML_METAL_EMBED_LIBRARY ON CACHE BOOL "" FORCE)
add_subdirectory("${WHISPER_SOURCE}" whisper-build)
add_executable(mimi-whisper-worker worker.cpp)
target_link_libraries(mimi-whisper-worker PRIVATE whisper)
target_compile_options(mimi-whisper-worker PRIVATE -Wall -Wextra -Werror)
